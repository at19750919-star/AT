    <script>
        // ========== 花色按鈕點擊事件 ==========
        document.querySelectorAll('.suit-button').forEach(button => {
            button.addEventListener('click', (e) => {
                e.preventDefault();
                button.classList.toggle('selected');
                // 更新訊號牌張數
                if (typeof window.updateSignalCardCount === 'function') {
                    window.updateSignalCardCount();
                }
            });
        });

        // ========== 數字牌按鈕點擊事件 ==========
        document.querySelectorAll('.rank-button').forEach(button => {
            button.addEventListener('click', (e) => {
                e.preventDefault();
                button.classList.toggle('selected');

                // 同步更新對應的 checkbox
                const value = button.dataset.value;
                const checkbox = document.querySelector(`.rank-checkbox[value="${value}"]`);
                if (checkbox) {
                    checkbox.checked = button.classList.contains('selected');
                    checkbox.dispatchEvent(new Event('change', { bubbles: true }));
                }
            });
        });

        // ========== 訊號牌快速設定 ==========
        function setSignalPreset(preset) {
            // 先清除所有選擇
            if (typeof window.clearSignalSelections === 'function') {
                window.clearSignalSelections();
            }
            let suits = [], ranks = [];
            if (preset === 'allA2') {
                suits = ['♠', '♥', '♦', '♣'];
                ranks = ['A', '2'];
            } else if (preset === 'hd0') {
                suits = ['♥', '♦'];
                ranks = ['10', 'J', 'Q', 'K'];
            }
            // 選花色
            document.querySelectorAll('.suit-button').forEach(btn => {
                if (suits.includes(btn.dataset.value)) btn.classList.add('selected');
            });
            // 選數字
            document.querySelectorAll('.rank-button').forEach(btn => {
                if (ranks.includes(btn.dataset.value)) {
                    btn.classList.add('selected');
                    const cb = document.querySelector(`.rank-checkbox[value="${btn.dataset.value}"]`);
                    if (cb) cb.checked = true;
                }
            });
            // 套用並更新顯示
            if (typeof window.updateSignalCardCount === 'function') window.updateSignalCardCount();
            if (typeof applySignalConfig === 'function') applySignalConfig();
            else if (typeof window.applySignalConfig === 'function') window.applySignalConfig();
        }

        // ========== 獲取選中的花色 ==========
        function getSelectedSuitsFromButtons() {
            const selectedButtons = document.querySelectorAll('.suit-button.selected');
            return Array.from(selectedButtons).map(btn => btn.dataset.value);
        }

        // ========== 更新訊號牌張數計算 ==========
        if (typeof window.updateSignalCardCount === 'undefined') {
            window.updateSignalCardCount = function () {
                const suits = getSelectedSuitsFromButtons();
                const ranks = Array.from(document.querySelectorAll('.rank-checkbox:checked')).map(cb => cb.value);
                const totalCards = suits.length * ranks.length * 8;
                const countElement = document.getElementById('signalCardCount');
                if (countElement) {
                    countElement.textContent = totalCards;
                }
                const configEl = document.getElementById('signalConfigDisplay');
                if (configEl && typeof formatSignalConfigText === 'function') {
                    configEl.textContent = formatSignalConfigText(suits, ranks);
                }
            };
        }

        // ========== 監聽 checkbox 變化 ==========
        document.querySelectorAll('.rank-checkbox').forEach(checkbox => {
            checkbox.addEventListener('change', () => {
                if (typeof window.updateSignalCardCount === 'function') {
                    window.updateSignalCardCount();
                }
            });
        });

        // ========== 違規統計初始化 ==========
        document.addEventListener('DOMContentLoaded', function () {
            const violationCards = document.querySelectorAll('.violation-card');
            violationCards.forEach(card => {
                card.classList.add('no-violation');
            });

            if (typeof updateViolationUI === 'function') {
                updateViolationUI(null);
            }

            // ===== 混和卡色模式勾選框 =====
            const mixedToggle = document.getElementById('mixedColorModeToggle');
            if (mixedToggle) {
                try {
                    const _mmLs = localStorage.getItem('card_color_mixed_mode');
                    mixedToggle.checked = (_mmLs === null) ? true : (_mmLs === '1');
                } catch (e) {
                    mixedToggle.checked = true;
                }
                mixedToggle.addEventListener('change', () => {
                    if (typeof persistCardColorMixedMode === 'function') {
                        persistCardColorMixedMode(mixedToggle.checked);
                    }
                    if (typeof log === 'function') {
                        log(
                            mixedToggle.checked
                                ? '卡色模式：混和（BBBR / RRRB / BBRR / RRBB）'
                                : '卡色模式：標準（BBBR / RRRB）',
                            'info'
                        );
                    }
                });
            }
        });

    <!-- 設定持久化：儲存/讀取上次的設定 -->
    <script>
        (function(){
            var STORAGE_KEY = 'at-settings';
            var INPUT_IDS = ['avgRecoveryLimit','maxSideLimit','maxTieLimit','maxFourCardRate','max7PtReversal','swapBanker6Target'];
            var CHECK_IDS = ['skipBanker6'];

            // 讀取上次設定
            try {
                var saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
                INPUT_IDS.forEach(function(id){
                    var el = document.getElementById(id);
                    if (el && saved[id] !== undefined && saved[id] !== null) el.value = saved[id];
                });
                CHECK_IDS.forEach(function(id){
                    var el = document.getElementById(id);
                    if (el && saved[id] !== undefined) el.checked = !!saved[id];
                });
            } catch(e){}

            // 任一設定變更時自動儲存
            function saveAll(){
                var obj = {};
                INPUT_IDS.forEach(function(id){
                    var el = document.getElementById(id);
                    if (el) obj[id] = el.value;
                });
                CHECK_IDS.forEach(function(id){
                    var el = document.getElementById(id);
                    if (el) obj[id] = el.checked;
                });
                try { localStorage.setItem(STORAGE_KEY, JSON.stringify(obj)); } catch(e){}
            }
            INPUT_IDS.concat(CHECK_IDS).forEach(function(id){
                var el = document.getElementById(id);
                if (el) el.addEventListener('change', saveAll);
                if (el && el.type === 'number') el.addEventListener('input', saveAll);
            });
        })();
    </script>

    <!-- 切牌 / 定位：點按鈕才展開輸入框 -->
    <script>
        (function () {
            var card = document.querySelector('.cut-card');
            if (!card) return;
            var PAIRS = { btnCut: 'cutPos', btnHighlightCard: 'highlightCardPos' };
            // capture 階段攔截：輸入框還沒展開時，這一下只負責展開，不執行切牌/定位
            card.addEventListener('click', function (e) {
                var btn = e.target.closest('button');
                if (!btn || !PAIRS[btn.id]) return;
                var input = document.getElementById(PAIRS[btn.id]);
                if (input && !input.classList.contains('is-open')) {
                    e.stopPropagation();
                    input.classList.add('is-open');
                    input.focus();
                }
            }, true);
            Object.keys(PAIRS).forEach(function (btnId) {
                var input = document.getElementById(PAIRS[btnId]);
                if (!input) return;
                input.addEventListener('keydown', function (e) {
                    if (e.key === 'Enter') document.getElementById(btnId).click();
                });
                // 沒填東西就移開焦點 → 收合（點自己那顆按鈕不算）
                input.addEventListener('blur', function (e) {
                    if (e.relatedTarget && e.relatedTarget.id === btnId) return;
                    if (!input.value) input.classList.remove('is-open');
                });
            });
        })();
    </script>
